**Q04.** 다음 C언어 코드에서 알맞은 출력 값을 작성하시오.

```c
#include <stdio.h>
#define SIZE 3
typedef struct {
    int a[SIZE];
    int front;
    int rear;
} Queue;
void enq(Queue* q, int val) {
    q->a[q->rear] = val;
    q->rear = (q->rear + 1) % SIZE;
}
int deq(Queue* q) {
    int val = q->a[q->front];
    q->front = (q->front + 1) % SIZE;
    return val;
}
int main() {
    Queue q = {{0}, 0, 0};
    enq(&q, 1); enq(&q, 2); deq(&q); enq(&q, 3);
    int first = deq(&q);
    int second = deq(&q);
    printf("%d\n%d", first, second);
    return 0;
}
```
